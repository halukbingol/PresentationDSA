% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
\subsubsection{Uploading}
\label{sec:app:upload}

\begin{enumerate}[label=\roman*.]
	
	\item Right click on \hCode{src}.

	\item Select \menu{Show in > System Explorer} to open 
	\hCode{src} in the \hCode{System Explorer}.

	\item \textbf{Zipping.}
	Compress \hCode{src} directory in \emph{zip format}. 
	Make sure that the name of the compressed file is \hCode{src.zip}.

	\textbf{Warning.}
	\begin{itemize}
		
		\item Only \hCode{zip} compression is accepted. 
		Other compressions, such as \hCode{rar}, are not accepted.

		\item Do not zip
		\begin{itemize}[nosep] 

			\item workspace

			\item project

			\item package
			
		\end{itemize}
			
		\end{itemize}
		but \hCode{src} only.
		Note that \hCode{src} is marked in  \reffig{fig:DirectoryStructure}.

	\item \textbf{Check directory structure.}
	Check if \hCode{src.zip} is correct
	\begin{enumerate}
		
		\item Create a new folder with your name on Desktop.
		Lets call it as \hCode{<MyName>}.
		
		\item Copy \hCode{src.zip} into \hCode{<MyName>} 
		
		\item Expand \hCode{src.zip} in \hCode{<MyName>} 
		The directory structure given in \reffig{fig:DirectoryStructure} should be obtained.
		Otherwise AutoGrader will assign 0, 
		since it cannot locate your files to grade.
	\end{enumerate}


	\item Upload \hCode{src.zip} to \YULearn.
		
\end{enumerate}
