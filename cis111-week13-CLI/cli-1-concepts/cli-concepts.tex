% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ A




% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ V frm
\begin{frame}[t,fragile] \frametitle{From source code to a running program}
  \begin{center}
  \begin{tikzpicture}[node distance=1.5cm]
    \node[file] (src) {Hello.java\\ \scriptsize source};
    \node[file, right=1.25cm of src] (cls) {Hello.class\\ \scriptsize bytecode};
    \node[file, right=1.25cm of cls] (jar) {hello.jar\\ \scriptsize archive};
    \node[box, fill=green!10, right=1.25cm of jar, minimum height=0.8cm, minimum width=2.2cm] (jvm)
         {\textbf{JVM}\\ \scriptsize runs bytecode};
    \draw[flow] (src) -- node[above, tool]{javac} node[below, lbl]{compile} (cls);
    \draw[flow] (cls) -- node[above, tool]{jar} node[below, lbl]{package} (jar);
    \draw[flow] (jar) -- node[above, tool]{java} node[below, lbl]{run} (jvm);
    \draw[flow, dashed] (cls.south) to[bend right=22] node[below, lbl]{or run the classes directly} (jvm.south);
    \foreach \o/\x in {Windows/-3.4, macOS/-1.5, Linux/0.4}
      \node[box, fill=gray!10, font=\scriptsize, minimum width=1.7cm] (\o) at ($(jvm.north)+(\x,1.25)$) {\o};
    \foreach \o in {Windows, macOS, Linux} \draw[thick, gray] (jvm.north) -- (\o.south);
  \end{tikzpicture}
  \end{center}
  \vspace{-0.2em}
  \begin{itemize}\small
    \item \textbf{Compile once}: \texttt{javac} turns Java source into \textbf{bytecode}, instructions for a virtual machine, not for a real CPU
    \item \textbf{Run anywhere}: every operating system has its own JVM that executes the same \texttt{.class} and \texttt{.jar} files
    \item The command line is where all of this happens; IDEs (Eclipse, IntelliJ, VS Code) run the same tools for you
  \end{itemize}
\end{frame}

        \node[box, fill=green!12, minimum width=5.4cm, minimum height=1.3cm, align=center, font=\scriptsize] at (0,-1.35)

    \begin{column}{0.46\textwidth}
      \begin{itemize}\small
        \item \textbf{JVM}: the virtual machine that executes bytecode
        \item \textbf{Runtime} (JRE): JVM + standard library + the \texttt{java} command; enough to \emph{run}
        \item \textbf{JDK}: runtime + development tools; needed to \emph{compile and package}
      \end{itemize}
      \begin{block}{Install a JDK}
        \small Students need the JDK. Since Java 11 Oracle no longer ships a separate JRE; some vendors still offer
        JRE packages, and \texttt{jlink} builds a minimal runtime for one application.
      \end{block}
    \end{column}
  \end{columns}

\begin{frame}[t,fragile] \frametitle{The main tools}
  \footnotesize
  \begin{tabular}{@{}L{0.13\textwidth}L{0.37\textwidth}L{0.44\textwidth}@{}}
    \toprule
    \textbf{Tool} & \textbf{Does} & \textbf{Typical use} \\
    \midrule
    \texttt{javac}     & compiles \texttt{.java} to \texttt{.class}       & \texttt{javac -d classes Hello.java} \\
    \texttt{java}      & starts a JVM and runs a program             & \texttt{java -cp classes Hello}, \texttt{java -jar app.jar} \\
    \texttt{jar}       & creates and lists JAR archives              & \texttt{jar --create --file app.jar -C classes .} \\
    \texttt{javadoc}   & HTML documentation from comments            & \texttt{javadoc -d docs src/*.java} \\
    \texttt{jshell}    & interactive Java (read--eval--print loop)  & trying out expressions \\
    \texttt{javap}     & disassembles \texttt{.class} files          & \texttt{javap -c Hello} \\
    \texttt{jdeps}     & which packages and modules a JAR needs      & \texttt{jdeps app.jar} \\
    \texttt{jlink}     & builds a custom, smaller runtime            & \texttt{jlink --add-modules java.base \dots} \\
    \texttt{jpackage}  & native installer (\texttt{.exe}, \texttt{.msi}, \texttt{.dmg}, \texttt{.pkg}, \dots) & shipping to end users \\
    \texttt{jcmd}, \texttt{jfr} & inspects running JVMs              & diagnostics, profiling \\
    \bottomrule
  \end{tabular}
\end{frame}

\begin{frame}[t,fragile] \frametitle{How the shell finds \texttt{javac}: \texttt{PATH}}
  \begin{center}
  \begin{tikzpicture}[x=1cm, y=1cm]
    \node[box, fill=gray!10, font=\ttfamily\small] (cmd) at (0,0) {\$ javac Hello.java};
    \node[lbl, above=2pt of cmd] {you type};
    \node[font=\small, anchor=west] at (2.2,0) {the shell looks in each \texttt{PATH} folder, in order:};
    \node[box, font=\ttfamily\scriptsize, minimum width=6.2cm, anchor=west] at (2.6,-0.9) {/usr/local/bin};
    \node[font=\scriptsize, anchor=west] at (8.9,-0.9) {no \texttt{javac} here};
    \node[box, font=\ttfamily\scriptsize, minimum width=6.2cm, anchor=west] at (2.6,-1.6) {/usr/bin};
    \node[font=\scriptsize, anchor=west] at (8.9,-1.6) {no \texttt{javac} here};
    \node[box, fill=green!12, font=\ttfamily\scriptsize, minimum width=6.2cm, anchor=west] at (2.6,-2.3) {/Library/Java/JavaVirtualMachines/\dots/bin};
    \node[font=\scriptsize\bfseries, anchor=west] at (8.9,-2.3) {found: run it};
    \draw[flow] (cmd.south) |- (2.6,-2.3);
  \end{tikzpicture}
  \end{center}
  \begin{itemize}\small
    \item \texttt{PATH} is an \textbf{environment variable}: a list of folders, searched from first to last
    \item The first match wins, so with two JDKs installed, \textbf{order matters}
    \item Not found anywhere: \win\ \texttt{'javac' is not recognized \dots}, \mac\ \texttt{zsh: command not found: javac}
    \item \texttt{JAVA\_HOME} is another variable: the JDK folder itself. Build tools (Maven, Gradle) and IDEs use it
  \end{itemize}
\end{frame}
% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ A frm




% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ V frm
\begin{frame}[t,fragile] \frametitle{The same ideas, different syntax}
  \small
  \begin{tabular}{@{}L{0.26\textwidth}L{0.33\textwidth}L{0.33\textwidth}@{}}
    \toprule
                            & \textbf{Windows (Command Prompt)}       & \textbf{macOS (Terminal, zsh)} \\
    \midrule
    show a variable         & \texttt{echo \%JAVA\_HOME\%}           & \texttt{echo \$JAVA\_HOME} \\
    where is a command      & \texttt{where javac}                   & \texttt{which javac} \\
    folder separator        & \texttt{\textbackslash}: \texttt{src\textbackslash Hello.java} & \texttt{/}: \texttt{src/Hello.java} \\
    path list separator     & \texttt{;} \quad (\texttt{-cp classes;lib\textbackslash a.jar}) & \texttt{:} \quad (\texttt{-cp classes:lib/a.jar}) \\
    list files              & \texttt{dir}                           & \texttt{ls} \\
    current folder          & \texttt{cd}                            & \texttt{pwd} \\
    last exit code          & \texttt{echo \%ERRORLEVEL\%}           & \texttt{echo \$?} \\
    set for this session    & \texttt{set NAME=value}                & \texttt{export NAME=value} \\
    set permanently         & \emph{System Properties, Environment Variables} & a line in \texttt{\textasciitilde/.zshrc} \\
    \bottomrule
  \end{tabular}

  \vspace{0.6em}
  \texttt{java}, \texttt{javac} and \texttt{jar} accept \texttt{/} in file paths on Windows too; the \textbf{path list} separator
  (\texttt{;} versus \texttt{:}) is the difference that really matters.
\end{frame}

  \begin{center}

    \node[box, font=\ttfamily\small, minimum width=2.2cm] (j) at (1.6,-1.4) {lib/json.jar};

  \begin{itemize}\small
    \item The \textbf{class path} lists folders and JAR files: \texttt{-cp} (or \texttt{-classpath}, \texttt{--class-path})
    \item Default: the current folder \texttt{.} (unless the \texttt{CLASSPATH} variable is set; better leave it unset)
    \item \win\ separates entries with \texttt{;}, \mac\ (and Linux) with \texttt{:}
    \item \texttt{javac} uses a class path too, to find the classes your code uses
  \end{itemize}
\end{frame}

    \begin{column}{0.5\textwidth}
      \begin{itemize}\small
        \item One file instead of a folder tree: classes, resources (text, images), metadata
        \item \textbf{Compressed} like a ZIP; any ZIP tool can open it
        \item \texttt{Main-Class} makes it \textbf{executable}: \texttt{java -jar hello.jar}
        \item \texttt{Class-Path} names other JARs it needs, relative to the JAR's folder
        \item A \textbf{library} JAR has no \texttt{Main-Class}; programs put it on their class path
        \item A ``fat'' (uber) JAR copies all libraries inside: one file to ship
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

  \begin{center}
  \begin{tikzpicture}[x=1cm, y=1cm, opt/.style={box, minimum width=4.2cm, minimum height=2.3cm, text width=3.9cm}]
    \node[opt, fill=yellow!10] at (0,0) {\textbf{1. JAR}\\[2pt]\footnotesize \texttt{app.jar}\\ user installs a JDK or JRE, then \texttt{java -jar app.jar}};
    \node[opt, fill=orange!10] at (4.8,0) {\textbf{2. Runtime image}\\[2pt]\footnotesize \texttt{jlink}: app + only the modules it needs, its own \texttt{bin/java}};
    \node[opt, fill=green!10] at (9.6,0) {\textbf{3. Native installer}\\[2pt]\footnotesize \texttt{jpackage}: \texttt{.msi}/\texttt{.exe}, \texttt{.dmg}/\texttt{.pkg}, \texttt{.deb}/\texttt{.rpm}};
    \draw[flow] (2.15,0) -- (2.65,0);
    \draw[flow] (6.95,0) -- (7.45,0);
    \node[lbl] at (4.8,-1.55) {more work for the developer, less for the user $\longrightarrow$};
  \end{tikzpicture}
  \end{center}
  \begin{itemize}\small
    \item \textbf{JAR}: smallest, one file for every OS; needs Java on the user's computer
    \item \textbf{Runtime image}: no Java installation needed; one image per OS (built on that OS)
    \item \textbf{Installer}: feels like any other app (Start menu, Applications folder); \texttt{jpackage} must run on the target OS
  \end{itemize}
\end{frame}

\begin{frame}[t,fragile] \frametitle{Check yourself}
  \begin{enumerate}\small
    \item \hQuestion{Which do you need to compile Java programs: a JRE or a JDK?}
          \onslide<2->{\hRemark{A JDK: the JRE has no \texttt{javac}.}}
    \item \hQuestion{\texttt{javac -version} works, but in a new folder \texttt{java Hello} says ``Could not find or load main class''. Why?}
          \onslide<3->{\hRemark{There is no \texttt{Hello.class} on the class path (here: the current folder). Compile first or \texttt{cd}.}}
    \item \hQuestion{Where does \texttt{javac -d classes} put \texttt{package a.b; class C}?}
          \onslide<4->{\hRemark{\texttt{classes/a/b/C.class}; run it with \texttt{java -cp classes a.b.C}.}}
    \item \hQuestion{Write the class path ``folder \texttt{out} and \texttt{lib/x.jar}'' for Windows and for macOS.}
          \onslide<5->{\hRemark{\texttt{out;lib\textbackslash x.jar} and \texttt{out:lib/x.jar}.}}
    \item \hQuestion{What makes \texttt{java -jar app.jar} know which class to start?}
          \onslide<6->{\hRemark{The \texttt{Main-Class} line in \texttt{META-INF/MANIFEST.MF}.}}
    \item \hQuestion{Your users have no Java installed. Which delivery options remain?}
          \onslide<7->{\hRemark{A \texttt{jlink} runtime image or a \texttt{jpackage} installer.}}
  \end{enumerate}
\end{frame}
