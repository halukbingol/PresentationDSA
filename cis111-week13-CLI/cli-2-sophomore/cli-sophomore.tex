\subsection[Encoding]{Source encoding}
% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ A




% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ V frm
\begin{frame}[t,fragile] \frametitle{Source file encoding: \hCode{-encoding}}
  \lstterminal[basicstyle=\ttfamily\tiny]{codeoutput/Encoding.txt}

  \vspace{0.1em}
  \small
  The file was saved in the old Turkish encoding ISO-8859-9. Since Java 18 \hCode{javac} assumes \textbf{UTF-8} on every system, so
  it rejects the bytes until we name the encoding, or convert the file (\hCode{iconv}; on \win{} use the editor's \emph{Save with encoding}). Older JDKs on Windows assumed Windows-1254 instead: the classic
  ``works on my machine'' bug. \hRemark{Save sources as UTF-8 and pass} \hCode{-encoding UTF-8} \hRemark{in builds.}
\end{frame}

\subsection[Release]{Target versions}

\subsection[Order]{Order, wildcards, CLASSPATH}
% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ A




% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ V frm
\begin{frame}[t,fragile] \frametitle{Order, wildcards and the \hCode{CLASSPATH} variable}
  \begin{columns}[T]
    \begin{column}{0.34\textwidth}
      \lstcode[basicstyle=\ttfamily\tiny]{codeoutput/ClassPath/App.java}{Java}
    \end{column}
    \begin{column}{0.64\textwidth}
      \uncover<2->{\lstterminal[basicstyle=\ttfamily\tiny]{codeoutput/ClassPath.txt}}
    \end{column}
  \end{columns}

  \vspace{0.1em}
  \small Two JARs with the same class: the \textbf{first} on the class path wins. \hCode{lib/*} means every JAR in \hCode{lib}
  (quote it on macOS; Windows: \hCode{-cp ".;lib\textbackslash *"}). A \hCode{CLASSPATH} variable \textbf{replaces} the default \hCode{.},
  so \hCode{App} itself disappears; prefer \hCode{-cp} and leave \hCode{CLASSPATH} unset.
\end{frame}

\subsection[Fat JAR]{Fat JARs}
% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ A




% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ V frm
\begin{frame}[t,fragile] \frametitle{A ``fat JAR: everything in one file}
  \lstterminal[basicstyle=\ttfamily\tiny]{codeoutput/FatJar.txt}

  \vspace{0.1em}
  \small
  One file is easy to ship, but you repackage other people's code (check licences), files with the same name collide,
  and signed library JARs break. Build tools automate it (Maven Shade, Gradle Shadow).

  \begin{columns}[T]
    \begin{column}{0.5\textwidth}


% +++++++++++++++++++++++++++++++++++++++ V lst
%: -Lst.
\begin{lstlisting}[style=codesnippet, language=XML, basicstyle=\ttfamily\tiny]
<project>
  <modelVersion>4.0.0</modelVersion>
  <groupId>edu.yeditepe</groupId>
  <artifactId>books</artifactId>
  <version>1.0</version>
  <properties>
    <maven.compiler.release>21</maven.compiler.release>
    <project.build.sourceEncoding>UTF-8</project.build.sourceEncoding>
  </properties>
  <dependencies>
    <dependency>
      <groupId>edu.yeditepe</groupId>
      <artifactId>table</artifactId>
      <version>1.0</version>
    </dependency>
  </dependencies>
</project>
\end{lstlisting}
% +++++++++++++++++++++++++++++++++++++++ A lst


    \end{column}
    \begin{column}{0.46\textwidth}
      \small \hCode{pom.xml} \emph{describes} the project; Maven knows the steps:
      \vspace{0.2em}

      \footnotesize
      \begin{tabular}{@{}ll@{}}
        \toprule
        \textbf{Command} & \textbf{By hand} \\
        \midrule
        \hCode{mvn compile}  & \hCode{javac -d \dots} \\
        \hCode{mvn test}     & run the tests \\
        \hCode{mvn package}  & \hCode{jar --create \dots} \\
        \hCode{mvn clean}    & \hCode{rm -rf target} \\
        \bottomrule
      \end{tabular}

      \vspace{0.3em}
      \small Sources go in \hCode{src/main/java}, resources in \hCode{src/main/resources}; libraries are downloaded from
      Maven Central. Same commands on Windows and macOS.
      \vspace{0.2em}

      \footnotesize\hRemark{Not run here}: Maven needs the internet, which these examples could not reach.
    \end{column}
  \end{columns}
\end{frame}

// build.gradle.kts

    application
}

java {
    toolchain { languageVersion = JavaLanguageVersion.of(21) }
}

    implementation(files("lib/table.jar"))
}

    mainClass = "edu.yeditepe.books.Main"

        \hCode{./gradlew run}     & \hCode{gradlew run} \\
        \hCode{./gradlew jar}     & \hCode{gradlew jar} \\
        \bottomrule
      \end{tabular}

      \vspace{0.4em}
      \small \hCode{gradlew} (the \emph{wrapper}) downloads the right Gradle version, so everyone builds the same way.
      The \hCode{application} plugin also makes start scripts for Windows and macOS.

\begin{frame}[t,fragile] \frametitle{Recipes}
  \footnotesize
  \begin{tabular}{@{}L{0.36\textwidth}L{0.58\textwidth}@{}}
    \toprule
    \textbf{Goal} & \textbf{Command (macOS; Windows: \hCode{;} and \hCode{\textbackslash})} \\
    \midrule
    compile a whole project         & \hCode{javac -d build -sourcepath src -encoding UTF-8 src/a/b/Main.java} \\
    \dots\ against a library        & \hCode{javac -d build -cp lib/x.jar -sourcepath src src/a/b/Main.java} \\
    \dots\ for an older Java        & \hCode{javac --release 17 \dots} \\
    with all warnings               & \hCode{javac -Xlint:all -Werror \dots} \\
    library JAR                     & \hCode{jar --create --file lib/x.jar -C build .} \\
    executable JAR + library        & \hCode{jar --create --file app.jar --manifest m.txt -C build . -C res .} \\
    run with libraries              & \hCode{java -cp "build:res:lib/*" a.b.Main} \\
    run tests with assertions       & \hCode{java -ea -cp build:test Tests} \\
    documentation                   & \hCode{javadoc -d docs -sourcepath src a.b} \\
    \bottomrule
  \end{tabular}

\begin{frame}[t,fragile] \frametitle{Exercises}
  \begin{enumerate}\small
    \item Add a column ``Pages'' to \hCode{books.csv} and print it; rebuild with \hCode{build.sh} or \hCode{build.bat}.
    \item Fix the padding bug in \hCode{Table.render} so that \hCode{TableTest} passes with \hCode{-ea}.

  \end{enumerate}
