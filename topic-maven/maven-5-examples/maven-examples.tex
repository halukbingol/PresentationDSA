\hSection[Lifecycle]{Phases and Goals}
% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ A




% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ V frm
\begin{frame}[t,fragile] \frametitle{A phase runs everything before it}
  \lstterminal[basicstyle=\ttfamily\tiny]{codeoutput/Phases.txt}

  \vspace{0.1em}
  \small
  \texttt{validate}: no goal bound for a jar project. \texttt{compile}: 2 goals. \texttt{package}: 6 goals, always in the same order.
\end{frame}
% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ A frm




% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ V frm
\begin{frame}[t,fragile] \frametitle{A goal runs alone}
  \lstterminal[basicstyle=\ttfamily\scriptsize]{codeoutput/Goal.txt}

  \vspace{0.1em}
  \small
  \texttt{jar:jar} without \texttt{compile}: nothing was compiled, so the jar has no class at all. Phases keep the order right; call goals directly only when you know what is already done.
\end{frame}
% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ A frm




% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ V frm
\begin{frame}[t,fragile] \frametitle{A failure stops the run}
  \lstterminal[basicstyle=\ttfamily\tiny]{codeoutput/Stop.txt}

  \vspace{0.1em}
  \small
  The test fails, so \texttt{jar:jar} never runs: no \texttt{.jar} in \texttt{target}. An untested jar cannot leave a Maven build by accident.
\end{frame}

\begin{frame}[t,fragile] \frametitle{Exercises}
  \begin{enumerate}\small
    \item Extend \texttt{ByHand} with a JUnit test, compiled and run by hand with the console launcher; compare with \texttt{mvn test}.
    \item Install \texttt{hello} as version \texttt{1.1}; what is added in \texttt{\textasciitilde/.m2/repository}?
    \item In \texttt{Scopes}, change JUnit to \texttt{compile} scope; how many jars does running need now?
    \item Run \texttt{mvn test} in \texttt{Phases}: which goals run?
    \item In \texttt{Inherit}, override \texttt{course} in the child and add \texttt{\$\{project.version\}} to the file.
  \end{enumerate}
\end{frame}
