\hSection[Idea]{The Idea}
% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ A




% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ V frm
\begin{frame}[t,fragile] \frametitle{A pattern describes a set of strings}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}\small
        \item A \textbf{regular expression} (regex) is a small formula for text.
        \item It stands for a \hEmph{set} of strings: ``four digits'', ``a word with a capital first letter'', ``a Yeditepe student e-mail''.
        \item A string \textbf{matches} the regex if it is in the set.
        \item One line of regex replaces a loop with many \hCode{if}s.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{center}\small
      \begin{tabular}{@{}lll@{}}
        \toprule
        \textbf{regex}                       & \textbf{matches}  & \textbf{does not} \\
        \midrule
        \hCode{cat}                         & cat               & Cat, cats \\
        \hCode{c.t}                         & cat, cut, c9t     & ct, cart \\
        \hCode{ca*t}                        & ct, cat, caaat    & cut \\
        \hCode{[0-9]\{4\}}                  & 2026, 0042        & 202, 20a6 \\
        \hCode{cat|dog}                     & cat, dog          & cow \\
        \bottomrule
      \end{tabular}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t,fragile] \frametitle{Repetition, position, grouping}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{center}\small
      \begin{tabular}{@{}ll@{}}
        \toprule
        \textbf{quantifier}      & \textbf{how many times} \\
        \midrule
        \hCode{x?}              & 0 or 1 \\
        \hCode{x*}              & 0 or more \\
        \hCode{x+}              & 1 or more \\
        \hCode{x\{3\}}          & exactly 3 \\
        \hCode{x\{2,4\}}        & 2 to 4 \\
        \hCode{x\{2,\}}         & 2 or more \\
        \bottomrule
      \end{tabular}
      \end{center}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{center}\small
      \begin{tabular}{@{}ll@{}}
        \toprule
        \textbf{regex}                & \textbf{meaning} \\
        \midrule
        \hCode{\textasciicircum}     & start of the text \\
        \hCode{\$}                   & end of the text \\
        \hCode{\textbackslash b}     & a word boundary \\
        \hCode{(\dots)}              & a group: one unit, remembered \\
        \hCode{x|y}                  & x or y \\
        \bottomrule
      \end{tabular}
      \end{center}
    \end{column}
  \end{columns}
  \vspace{0.5em}
  \begin{itemize}\small
    \item A quantifier applies to the one thing before it: \hCode{ab+} is a, b, b, \dots; \hCode{(ab)+} is ab, abab, \dots
    \item \hCode{|} binds weakest: \hCode{\textasciicircum cat|dog\$} is ``\hCode{\textasciicircum cat}'' or ``\hCode{dog\$}''; write \hCode{\textasciicircum (cat|dog)\$}.
  \end{itemize}
\end{frame}

  \begin{tabular}{@{}cccccccc@{}}

  \end{itemize}

\hSection[Engines]{How Matching Works}
% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ A




% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ V frm
\begin{frame}[t,fragile] \frametitle{Idea 1: a finite automaton}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \begin{center}
      \begin{tikzpicture}[>=stealth, node distance=2.2cm, auto, every state/.style={minimum size=0.8cm, fill=blue!8}]
        \node[state, initial]   (q0)               {$q_{0}$};
        \node[state]            (q1) [right of=q0] {$q_{1}$};
        \node[state, accepting] (q2) [right of=q1] {$q_{2}$};
        \path[->] (q0) edge node {c} (q1)
                  (q1) edge [loop above] node {a} (q1)
                       edge node {t} (q2);
      \end{tikzpicture}
      \end{center}
      \small An automaton for \hCode{ca*t}: \hCode{ct}, \hCode{cat}, \hCode{caaat} end in $q_{2}$; \hCode{ca} stops in $q_{1}$, \hCode{cut} has no arrow for \hCode{u}.
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{itemize}\small
        \item Every regex (without the extras) can be turned into a machine with finitely many \emph{states}.
        \item Read the text one character at a time and follow the arrow.
        \item At the end: in an accepting (double) state $\Rightarrow$ match.
        \item Time grows only linearly with the text.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ V frm
\begin{frame}[t,fragile] \frametitle{Idea 2: try, and go back (backtracking)}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}


% +++++++++++++++++++++++++++++++++++++++ V alg
%: -Alg. 
\begin{algorithm}[H]
  \caption{Matching by trying}
  \KwIn{a regex as a list of parts $p_{1} \dots p_{k}$, a text $s$}
  \KwOut{true if $s$ matches}
  start with part $p_{1}$ at position 1\;
  \While{a part is left}{
    \eIf{the part can take a (new) choice here}{
      take it, move on to the next part\;
    }{
      go back to the last part that had another choice\;
      \lIf{there is none}{\Return{false}}
    }
  }
  \Return{true if the whole text is used}
\end{algorithm}
% +++++++++++++++++++++++++++++++++++++++ A alg

    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{itemize}\small
        \item Java, Python, JavaScript, Perl work this way.
        \item It allows extras: back-references \hCode{\textbackslash 1}, look-ahead.
        \item \hCode{a*} first takes as many as it can (\emph{greedy}), then gives back one at a time.
        \item Too many choices to go back to: the time can explode (senior level).
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ A frm




% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ V frm
\begin{frame}[t,fragile] \frametitle{Greedy, lazy and the cost of choices}
  \begin{itemize}\small
    \item Text \hCode{<b>Bold</b> and <b>more</b>}, regex \hCode{<b>.*</b>}:
      \hCode{.*} takes everything, gives back until \hCode{</b>} fits: the match runs to the \emph{last} \hCode{</b>}.
    \item Lazy \hCode{<b>.*?</b>}: takes as little as possible: \hCode{<b>Bold</b>}.
    \item Exact \hCode{<b>[\textasciicircum <]*</b>}: only one way to match, so nothing to try again.
    \item With $n$ characters and two ways to split each, a backtracking engine may try up to
  \end{itemize}
  \[
  	2^{n} \ \mathrm{paths}, \qquad 2^{30} \approx 10^{9}
  \]
  \begin{itemize}\small
    \item Good patterns leave few choices: say exactly what may appear.
  \end{itemize}
\end{frame}

\hSection[Limits]{Limits}
% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ A




% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ V frm
\begin{frame}[t,fragile] \frametitle{What a regex cannot (or should not) do}
  \begin{itemize}\small
    \item \textbf{Counting without a bound}: ``as many \hCode{(} as \hCode{)}'' is not regular; brackets, HTML, JSON need a parser.
    \item \textbf{Meaning}: \hCode{\textbackslash d\{2\}\textbackslash .\textbackslash d\{2\}\textbackslash .\textbackslash d\{4\}} accepts \hCode{31.02.2026}; the calendar check needs code (\hCode{LocalDate.parse}).
    \item \textbf{Readability}: \hCode{\textasciicircum (?=.*\textbackslash d)(?=.*[a-z]).\{8,\}\$} is hard to read; comment it, name the groups, test it.
    \item \textbf{Speed}: some patterns take exponential time on unlucky input.
    \item \textbf{Security}: a pattern built from user input can do unwanted things.
  \end{itemize}
  \vspace{0.5em}
  \hRemark{A regex is a tool for the \emph{form} of text, not for its structure or meaning.}
\end{frame}

\hSection[Summary]{Summary}
% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ A




% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ V frm
\begin{frame}[t,fragile] \frametitle{Summary}
  \begin{itemize}\small
    \item A regex describes a set of strings; a text matches if it is in the set.
    \item Building blocks: literals, classes, quantifiers, anchors, groups, alternation.
    \item In Java: compile to a \hCode{Pattern}, ask a \hCode{Matcher}; write every \hCode{\textbackslash} twice in source.
    \item Engines: an automaton (linear, fewer features) or backtracking (more features, can be slow).
    \item Regexes check the form of text; use parsers and code for structure and meaning.
  \end{itemize}
\end{frame}
